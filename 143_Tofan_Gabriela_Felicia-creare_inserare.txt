-- ============================== CREARE TABELE ==============================
-- =========================== TABELE INDEPENDENTE ===========================

CREATE TABLE client (
    id_client NUMBER(13) CONSTRAINT pk_client PRIMARY KEY,
    nume VARCHAR2(50) NOT NULL,
    prenume VARCHAR2(50) NOT NULL,
    email VARCHAR2(50) CONSTRAINT uq_client_email UNIQUE,
    telefon VARCHAR2(15) NOT NULL
);

CREATE TABLE angajat (
    id_angajat NUMBER(13) CONSTRAINT pk_angajat PRIMARY KEY,
    nume VARCHAR2(50) NOT NULL,
    prenume VARCHAR2(50) NOT NULL,
    rol VARCHAR2(30)
);

CREATE TABLE locatie (
    id_locatie NUMBER(13) CONSTRAINT pk_locatie PRIMARY KEY,
    nume VARCHAR2(50) NOT NULL,
    adresa VARCHAR2(100)
);

CREATE TABLE reducere (
    id_reducere NUMBER(13) CONSTRAINT pk_reducere PRIMARY KEY,
    nume_campanie VARCHAR2(50) NOT NULL,
    procent NUMBER(3) CONSTRAINT ck_reducere_procent CHECK (procent > 0 AND procent <= 100)
);

CREATE TABLE furnizor (
    id_furnizor NUMBER(13) CONSTRAINT pk_furnizor PRIMARY KEY,
    nume_furnizor VARCHAR2(50) NOT NULL,
    telefon_contact VARCHAR2(15)
);

-- ============================ TABELE DEPENDENTE ============================

CREATE TABLE categorie (
    id_categorie NUMBER(13) CONSTRAINT pk_categorie PRIMARY KEY,
    denumire VARCHAR2(50) NOT NULL,
    id_categorie_parinte NUMBER(13) CONSTRAINT fk_categorie_parinte REFERENCES categorie(id_categorie)
);

CREATE TABLE abonament (
    id_abonament NUMBER(13) CONSTRAINT pk_abonament PRIMARY KEY,
    id_client NUMBER(13) NOT NULL,
    tip_abonament VARCHAR2(30),
    data_creare DATE DEFAULT SYSDATE NOT NULL,
    data_expirare DATE,
    CONSTRAINT fk_abonament_client FOREIGN KEY (id_client) REFERENCES client(id_client),
    CONSTRAINT uq_abonament_client UNIQUE (id_client)
);

CREATE TABLE echipament (
    id_echipament NUMBER(13) CONSTRAINT pk_echipament PRIMARY KEY,
    denumire VARCHAR2(50) NOT NULL,
    pret_zi NUMBER(5,2) NOT NULL,
    id_categorie NUMBER(13) CONSTRAINT fk_echipament_categorie REFERENCES categorie(id_categorie),
    id_locatie NUMBER(13) CONSTRAINT fk_echipament_locatie REFERENCES locatie(id_locatie),
    id_furnizor NUMBER(13) CONSTRAINT fk_echipament_furnizor REFERENCES furnizor(id_furnizor),
    CONSTRAINT ck_echipament_pret CHECK (pret_zi > 0)
);

CREATE TABLE inchiriere (
    id_inchiriere NUMBER(13) CONSTRAINT pk_inchiriere PRIMARY KEY,
    id_client NUMBER(13) CONSTRAINT fk_inchiriere_client REFERENCES client(id_client),
    id_angajat_preluare NUMBER(13) NOT NULL,
    id_angajat_returnare NUMBER(13),
    id_reducere NUMBER(13),
    data_inceput DATE NOT NULL,
    data_final DATE NOT NULL,
    CONSTRAINT fk_inchiriere_ang_preluare FOREIGN KEY (id_angajat_preluare) REFERENCES angajat(id_angajat),
    CONSTRAINT fk_inchiriere_ang_returnare FOREIGN KEY (id_angajat_returnare) REFERENCES angajat(id_angajat),
    CONSTRAINT fk_inchiriere_reducere FOREIGN KEY (id_reducere) REFERENCES reducere(id_reducere)
);

CREATE TABLE detalii (
    id_inchiriere NUMBER(13),
    id_echipament NUMBER(13),
    id_angajat NUMBER(13),
    pret_unitar NUMBER(7,2) NOT NULL,
    stare_obiect VARCHAR2(50),
    CONSTRAINT pk_detalii PRIMARY KEY (id_inchiriere, id_echipament),
    CONSTRAINT fk_detalii_inchiriere FOREIGN KEY (id_inchiriere) REFERENCES inchiriere(id_inchiriere),
    CONSTRAINT fk_detalii_echipament FOREIGN KEY (id_echipament) REFERENCES echipament(id_echipament),
    CONSTRAINT fk_detalii_angajat FOREIGN KEY (id_angajat) REFERENCES angajat(id_angajat)
);

CREATE TABLE fisa_tehnica (
    id_echipament NUMBER(13) CONSTRAINT pk_fisa_tehnica PRIMARY KEY,
    ultima_verificare DATE,
    detalii_uzura VARCHAR2(500),
    CONSTRAINT fk_fisa_tehnica_echipament FOREIGN KEY (id_echipament) REFERENCES echipament(id_echipament)
);

CREATE TABLE plata (
    id_plata NUMBER(13) CONSTRAINT pk_plata PRIMARY KEY,
    id_inchiriere NUMBER(13) NOT NULL,
    suma_plata NUMBER(7,2) NOT NULL,
    data_plata DATE,
    CONSTRAINT fk_plata_inchiriere FOREIGN KEY (id_inchiriere) REFERENCES inchiriere(id_inchiriere)
);
